# Personal Finance Tracker

A beginner-friendly Java 26 Swing application for tracking Indian Rupee income and expenses. Data is stored in `finance.db` using SQLite, and reports use JFreeChart.

## Requirements

- Java 26
- Maven 3.10 or newer

## Run

From this directory:

```text
mvn compile exec:java -Dexec.mainClass="financetracker.Main"
```

Build the runnable shaded JAR:

```text
mvn clean package
java -jar target/finance-tracker-1.0.0.jar
```

Load approximately thirty sample transactions from the command line:

```text
mvn compile exec:java -Dexec.mainClass="financetracker.SeedData"
```

The database file is created automatically beside the project when the application first starts. Amounts are positive whole numbers, stored as `long`, and displayed with Indian grouping and the `₹` symbol.

## Features

- Dashboard cards for income, expense, balance, current-month totals, and top spending categories
- Add, edit, delete, and filter transactions
- Monthly category and month-by-month summaries
- Pie, bar, and balance-trend charts
- CSV export for filtered transactions and summaries
- Friendly validation and empty-database handling

## Eight test cases

1. Save `2026-10-27`, `Income`, `Salary`, `75,000`: transaction saves and displays as `₹75,000`.
2. Save an empty amount: shows `Please enter the amount.`
3. Save amount `0`: shows the whole-number validation error.
4. Save amount `100,000,001`: shows the whole-number validation error.
5. Enter amount `12.50`: rejects decimals.
6. Enter a future date: shows a future-date error.
7. Leave category empty in the edit dialog: shows the category validation error.
8. Select a row, choose Delete, and cancel confirmation: record remains unchanged.

/** Builds the pie, bar, and line charts used by the reports tab. */
public class ChartService {
    /** Creates a spending-by-category pie chart. */
    public JFreeChart createPieChart(Map<String, Long> totals) {
        DefaultPieDataset<String> dataset = new DefaultPieDataset<>();
        totals.forEach(dataset::setValue);
        JFreeChart chart = ChartFactory.createPieChart("Spending by Category", dataset, true, true, false);
        chart.setBackgroundPaint(Color.WHITE);
        PiePlot<?> plot = (PiePlot<?>) chart.getPlot(); plot.setBackgroundPaint(Color.WHITE); plot.setOutlineVisible(false);
        plot.setLabelGenerator(new StandardPieSectionLabelGenerator("{0}: ₹{1}"));
        return chart;
    }

    /** Creates a monthly income versus expense bar chart. */
    public JFreeChart createBarChart(Map<YearMonth, long[]> monthly) {
        DefaultCategoryDataset dataset = new DefaultCategoryDataset();
        monthly.forEach((month, values) -> { dataset.addValue(values[0], "Income", month.toString()); dataset.addValue(values[1], "Expense", month.toString()); });
        JFreeChart chart = ChartFactory.createBarChart("Monthly Income vs Expense", "Month", "Amount (₹)", dataset);
        chart.setBackgroundPaint(Color.WHITE); CategoryPlot plot = (CategoryPlot) chart.getPlot(); plot.setBackgroundPaint(Color.WHITE); plot.setOutlineVisible(false);
        plot.getRangeAxis().setLabelFont(new Font("Segoe UI", Font.PLAIN, 12)); plot.getRenderer().setSeriesPaint(0, Theme.INCOME); plot.getRenderer().setSeriesPaint(1, Theme.EXPENSE);
        return chart;
    }

    /** Creates a line chart showing balance trend by month. */
    public JFreeChart createLineChart(Map<YearMonth, long[]> monthly) {
        DefaultCategoryDataset dataset = new DefaultCategoryDataset();
        long balance = 0;
        for (Map.Entry<YearMonth, long[]> entry : monthly.entrySet()) {
            balance += entry.getValue()[0] - entry.getValue()[1];
            dataset.addValue(balance, "Balance", entry.getKey().toString());
        }
        JFreeChart chart = ChartFactory.createLineChart("Balance Trend", "Month", "Amount (₹)", dataset);
        chart.setBackgroundPaint(Color.WHITE); CategoryPlot plot = (CategoryPlot) chart.getPlot(); plot.setBackgroundPaint(Color.WHITE); plot.setOutlineVisible(false); plot.getRenderer().setSeriesPaint(0, Theme.ACCENT);
        return chart;
    }
}

package financetracker;

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.List;

/** Exports transactions and summaries as UTF-8 CSV files. */
public final class CsvExporter {
    private CsvExporter() { }

    /** Writes transaction records to a CSV file. */
    public static void exportTransactions(File file, List<Transaction> transactions) throws IOException {
        try (Writer writer = new OutputStreamWriter(new FileOutputStream(file), StandardCharsets.UTF_8)) {
            writer.write("Date,Type,Category,Amount (INR),Description\n");
            for (Transaction t : transactions) writer.write(String.join(",", csv(t.getDate().toString()), csv(t.getType()),
                    csv(t.getCategory()), Long.toString(t.getAmount()), csv(t.getDescription())) + "\n");
        }
    }

    /** Writes a monthly summary to a CSV file. */
    public static void exportSummary(File file, String month, long income, long expense, long balance,
                                     List<String[]> categoryRows, List<String[]> monthlyRows) throws IOException {
        try (Writer writer = new OutputStreamWriter(new FileOutputStream(file), StandardCharsets.UTF_8)) {
            writer.write("Month," + csv(month) + "\nIncome," + income + "\nExpense," + expense + "\nBalance," + balance + "\n\n");
            writer.write("Category,Amount (INR)\n");
            for (String[] row : categoryRows) writer.write(csv(row[0]) + "," + row[1] + "\n");
            writer.write("\nMonth,Income (INR),Expense (INR),Balance (INR)\n");
            for (String[] row : monthlyRows) writer.write(csv(row[0]) + "," + row[1] + "," + row[2] + "," + row[3] + "\n");
        }
    }

    private static String csv(String value) { return "\"" + value.replace("\"", "\"\"") + "\""; }
}

package financetracker;

import java.text.DecimalFormat;
import java.text.DecimalFormatSymbols;
import java.util.Locale;

/** Formats whole Indian Rupee amounts consistently. */
public final class CurrencyUtil {
    private CurrencyUtil() { }

    /** Formats a signed amount using Indian digit grouping. */
    public static String formatINR(long amount) {
        boolean negative = amount < 0;
        String digits = Long.toString(Math.abs(amount));
        if (digits.length() <= 3) return (negative ? "-₹" : "₹") + digits;
        String last = digits.substring(digits.length() - 3);
        String prefix = digits.substring(0, digits.length() - 3);
        StringBuilder grouped = new StringBuilder();
        while (prefix.length() > 2) {
            grouped.insert(0, "," + prefix.substring(prefix.length() - 2));
            prefix = prefix.substring(0, prefix.length() - 2);
        }
        grouped.insert(0, prefix);
        return (negative ? "-₹" : "₹") + grouped + "," + last;
    }

    /** Returns an amount formatted without a currency symbol for chart values. */
    public static String formatNumber(long amount) {
        DecimalFormat format = new DecimalFormat("#,##0", DecimalFormatSymbols.getInstance(Locale.US));
        return format.format(amount);
    }
}

package financetracker;

import java.sql.*;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

/** Provides SQLite persistence with prepared statements for every query. */
public class Database {
    private static final String URL = "jdbc:sqlite:finance.db";

    /** Opens the database and creates its table when needed. */
    public Database() throws SQLException {
        try (Connection connection = connect(); Statement statement = connection.createStatement()) {
            statement.executeUpdate("""
                    CREATE TABLE IF NOT EXISTS transactions (
                        id INTEGER PRIMARY KEY AUTOINCREMENT,
                        date TEXT NOT NULL,
                        type TEXT NOT NULL CHECK(type IN ('Income','Expense')),
                        category TEXT NOT NULL,
                        amount INTEGER NOT NULL CHECK(amount > 0),
                        description TEXT
                    )
                    """);
        }
    }

    private Connection connect() throws SQLException {
        return DriverManager.getConnection(URL);
    }

    /** Inserts a transaction and returns its generated id. */
    public long add(Transaction transaction) throws SQLException {
        String sql = "INSERT INTO transactions(date,type,category,amount,description) VALUES(?,?,?,?,?)";
        try (Connection connection = connect(); PreparedStatement ps = connection.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {
            bind(ps, transaction);
            ps.executeUpdate();
            try (ResultSet keys = ps.getGeneratedKeys()) {
                if (keys.next()) return keys.getLong(1);
            }
        }
        throw new SQLException("The database did not return a transaction id.");
    }

    /** Updates an existing transaction. */
    public void update(Transaction transaction) throws SQLException {
        String sql = "UPDATE transactions SET date=?,type=?,category=?,amount=?,description=? WHERE id=?";
        try (Connection connection = connect(); PreparedStatement ps = connection.prepareStatement(sql)) {
            bind(ps, transaction);
            ps.setLong(6, transaction.getId());
            ps.executeUpdate();
        }
    }

    /** Deletes a transaction by id. */
    public void delete(long id) throws SQLException {
        try (Connection connection = connect(); PreparedStatement ps = connection.prepareStatement("DELETE FROM transactions WHERE id=?")) {
            ps.setLong(1, id);
            ps.executeUpdate();
        }
    }

    /** Queries transactions using optional inclusive date, category, and type filters. */
    public List<Transaction> query(LocalDate from, LocalDate to, String category, String type) throws SQLException {
        StringBuilder sql = new StringBuilder("SELECT id,date,type,category,amount,description FROM transactions WHERE 1=1");
        List<Object> parameters = new ArrayList<>();
        if (from != null) { sql.append(" AND date >= ?"); parameters.add(from.toString()); }
        if (to != null) { sql.append(" AND date <= ?"); parameters.add(to.toString()); }
        if (category != null && !category.isBlank() && !"All".equals(category)) { sql.append(" AND category = ?"); parameters.add(category); }
        if (type != null && !type.isBlank() && !"All".equals(type)) { sql.append(" AND type = ?"); parameters.add(type); }
        sql.append(" ORDER BY date DESC, id DESC");
        try (Connection connection = connect(); PreparedStatement ps = connection.prepareStatement(sql.toString())) {
            for (int i = 0; i < parameters.size(); i++) ps.setObject(i + 1, parameters.get(i));
            try (ResultSet rs = ps.executeQuery()) {
                List<Transaction> result = new ArrayList<>();
                while (rs.next()) result.add(read(rs));
                return result;
            }
        }
    }

    /** Queries all transactions. */
    public List<Transaction> all() throws SQLException { return query(null, null, "All", "All"); }

    private void bind(PreparedStatement ps, Transaction t) throws SQLException {
        ps.setString(1, t.getDate().toString());
        ps.setString(2, t.getType());
        ps.setString(3, t.getCategory());
        ps.setLong(4, t.getAmount());
        ps.setString(5, t.getDescription());
    }

    private Transaction read(ResultSet rs) throws SQLException {
        return new Transaction(rs.getLong("id"), LocalDate.parse(rs.getString("date")), rs.getString("type"),
                rs.getString("category"), rs.getLong("amount"), rs.getString("description"));
    }
}

package financetracker;

import java.time.YearMonth;
import java.util.*;
import java.util.stream.Collectors;

/** Calculates dashboard and report summaries from transactions. */
public class ReportService {
    /** Sums income for a list. */
    public long totalIncome(List<Transaction> transactions) { return totalByType(transactions, "Income"); }
    /** Sums expenses for a list. */
    public long totalExpense(List<Transaction> transactions) { return totalByType(transactions, "Expense"); }
    /** Returns income minus expense. */
    public long balance(List<Transaction> transactions) { return totalIncome(transactions) - totalExpense(transactions); }
    /** Returns totals by category for expenses. */
    public Map<String, Long> categoryTotals(List<Transaction> transactions) {
        return transactions.stream().filter(t -> "Expense".equals(t.getType()))
                .collect(Collectors.groupingBy(Transaction::getCategory, TreeMap::new, Collectors.summingLong(Transaction::getAmount)));
    }
    /** Returns the three largest spending categories. */
    public List<Map.Entry<String, Long>> topThreeCategories(List<Transaction> transactions) {
        return categoryTotals(transactions).entrySet().stream().sorted(Map.Entry.<String, Long>comparingByValue().reversed()).limit(3).toList();
    }
    /** Returns totals grouped by calendar month. */
    public Map<YearMonth, long[]> monthlySummary(List<Transaction> transactions) {
        Map<YearMonth, long[]> result = new TreeMap<>();
        for (Transaction t : transactions) {
            long[] totals = result.computeIfAbsent(YearMonth.from(t.getDate()), ignored -> new long[2]);
            totals["Income".equals(t.getType()) ? 0 : 1] += t.getAmount();
        }
        return result;
    }
    /** Returns totals by type. */
    private long totalByType(List<Transaction> transactions, String type) {
        return transactions.stream().filter(t -> type.equals(t.getType())).mapToLong(Transaction::getAmount).sum();
    }
}

package financetracker;

import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.time.format.ResolverStyle;

/** Validates user-entered transactions and filter values. */
public final class Validator {
    private static final DateTimeFormatter DATE_FORMAT =
            DateTimeFormatter.ofPattern("uuuu-MM-dd").withResolverStyle(ResolverStyle.STRICT);
    private Validator() { }

    /** Parses a valid, non-future ISO date or throws a user-friendly error. */
    public static LocalDate parseDate(String value) {
        try {
            LocalDate date = LocalDate.parse(value.trim(), DATE_FORMAT);
            if (date.isAfter(LocalDate.now())) throw new IllegalArgumentException("Date cannot be in the future.");
            return date;
        } catch (DateTimeParseException | NullPointerException ex) {
            throw new IllegalArgumentException("Please enter a valid date (yyyy-MM-dd).");
        }
    }

    /** Parses a positive whole-number amount after removing separators. */
    public static long parseAmount(String value) {
        if (value == null || value.trim().isEmpty()) throw new IllegalArgumentException("Please enter the amount.");
        String normalized = value.replace(",", "").replace(" ", "").trim();
        if (!normalized.matches("\\d+")) throw new IllegalArgumentException("Amount must be a whole number greater than 0.");
        try {
            long amount = Long.parseLong(normalized);
            if (amount <= 0 || amount > 100_000_000L) throw new NumberFormatException();
            return amount;
        } catch (NumberFormatException ex) {
            throw new IllegalArgumentException("Amount must be a whole number greater than 0.");
        }
    }

    /** Validates the remaining transaction fields. */
    public static void validateTransaction(LocalDate date, String type, String category) {
        if (date == null || date.isAfter(LocalDate.now())) throw new IllegalArgumentException("Date cannot be in the future.");
        if (!"Income".equals(type) && !"Expense".equals(type)) throw new IllegalArgumentException("Type must be Income or Expense.");
        if (category == null || category.trim().isEmpty()) throw new IllegalArgumentException("Category must not be empty.");
    }
}
